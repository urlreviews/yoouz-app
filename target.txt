                    >
                      {/* Video Thumbnail / Preview */}

                            alt={video.placeName || "Review"}

                            referrerPolicy="no-referrer"

                          />

                            <Video className="w-6 h-6" />

                          <div className="w-8 h-8 rounded-full bg-white/90 shadow-md flex items-center justify-center text-zinc-900 group-hover:scale-110 transition-transform">
                            <Play className="w-4 h-4 fill-current ml-0.5" />
                          </div>
                        </div>
                        {/* Rating Badge on Thumbnail */}
                        <div className="absolute top-1.5 left-1.5 px-1.5 py-0.5 rounded-md bg-black/70 backdrop-blur-xs text-[10px] font-bold text-amber-300 flex items-center gap-0.5">

                          <span>{video.rating || 5}</span>

                      {/* Video Info */}
                      <div className="flex-1 min-w-0 flex flex-col justify-between py-0.5">

                            <MapPin className="w-3 h-3 shrink-0" />

                          <p className="text-xs font-bold text-zinc-900 line-clamp-2 leading-snug group-hover:text-[#1a73e8] transition-colors">
                            {video.caption || `${video.placeName || "Authentic Review"} - Verified Customer Experience`}
                          </p>
                          {video.dishOrItem && (
                            <span className="inline-block text-[10px] font-semibold bg-zinc-100 text-zinc-600 px-2 py-0.5 rounded-md">
                              {video.dishOrItem}
                            </span>
                          )}
                        </div>

                        <div className="flex items-center justify-between text-[11px] text-zinc-500 pt-2 border-t border-zinc-100">

                            <span className="flex items-center gap-1 font-semibold">
                              <ThumbsUp className="w-3 h-3 text-zinc-400" />
                              <span>{video.likes || 0}</span>
                            </span>
                            <span>•</span>

                          <span className="text-[11px] font-bold text-[#1a73e8] group-hover:underline">
                            Watch Review
                          </span>
                        </div>
                      </div>
                    </div>
                  );
                })}
              </div>
            ) : (
              <div className="bg-white rounded-2xl border border-zinc-200 p-6 text-center shadow-xs space-y-2">
                <div className="w-10 h-10 rounded-full bg-zinc-100 text-zinc-400 flex items-center justify-center mx-auto">
                  <Video className="w-5 h-5" />
                </div>
                <p className="text-xs font-bold text-zinc-700">No Video Reviews Yet</p>
                <p className="text-[11px] text-zinc-400 max-w-[200px] mx-auto">
                  {isOwner 
                    ? "Record your first 60-second authentic video review to help the community." 
                    : `When ${author.name || "this reviewer"} shares video reviews, they will appear here.`}
                </p>
                {isOwner && onRecordReview && (
                  <button
                    type="button"
                    onClick={() => onRecordReview(null)}
                    className="mt-2 inline-flex items-center gap-1.5 px-4 py-2 rounded-full bg-[#1a73e8] text-white text-xs font-bold hover:bg-[#1557b0] transition-colors cursor-pointer shadow-xs"
                  >
                    <Video className="w-3.5 h-3.5" />
                    <span>Record First Review</span>
                  </button>

            )}
